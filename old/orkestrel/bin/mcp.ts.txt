// === TUI Rendering

/** Render polished startup banner to stderr */
function renderBanner(
	output: WritableTarget,
	options: {
		port: number
		tools: readonly ToolInterface[]
	},
): void {
	const toolNames = options.tools.map((t) => t.name).join(LABEL_DIM(' · '))

	const lines = [
		`${LABEL_CYAN(LABEL_BOLD('orkestrel'))} ${LABEL_DIM('v0.0.1')}`,
		'',
		`${LABEL_DIM('stdio')}   JSON-RPC`,
		`${LABEL_DIM('http')}    ${LABEL_CYAN(`http://localhost:${options.port}/mcp`)}`,
		`${LABEL_DIM('ws')}      ${LABEL_CYAN(`ws://localhost:${options.port}/mcp`)}`,
		`${LABEL_DIM('prompt')}  ${LABEL_CYAN(`http://localhost:${options.port}/api/prompts`)}`,
		'',
		`${LABEL_DIM('tools')}   ${toolNames}`,
		'',
		`${ICON_CHECK} ${LABEL_GREEN(LABEL_BOLD('Ready'))}`,
	]

	const box = renderBox(lines.join('\n'), BOX_ROUND, 2, undefined, undefined)
	output.write('\n')
	for (const line of box) {
		output.write(`  ${line}\n`)
	}
	output.write('\n')
}

/** Render polished companion startup banner */
function renderCompanionBanner(output: WritableTarget, url: string): void {
	const lines = [
		`${LABEL_CYAN(LABEL_BOLD('orkestrel'))} ${LABEL_DIM('companion')}`,
		'',
		`${LABEL_DIM('Connecting to')} ${LABEL_CYAN(url)}`,
	]

	const box = renderBox(lines.join('\n'), BOX_ROUND, 2, undefined, undefined)
	output.write('\n')
	for (const line of box) {
		output.write(`  ${line}\n`)
	}
	output.write('\n')
}

	// Track active SSE connections for shutdown broadcast

	/** Broadcast a shutdown event to all connected SSE clients */
	function broadcastShutdown(): void {
		for (const res of sseConnections) {
			try {
				res.write('event: shutdown\ndata: {}\n\n')
			} catch {
				// Connection may already be closed
			}
		}
		sseConnections.clear()
	}

	// SSE endpoint: GET /api/prompts — stream pending prompts
	httpServer.routes.register({
		method: 'GET',
		path: '/api/prompts',
		handler({ request: req, response: res }: RouteHandlerContext) {
			res.writeHead(200, {
				'Content-Type': 'text/event-stream',
				'Cache-Control': 'no-cache',
				Connection: 'keep-alive',
			})

			sseConnections.add(res)
			let closed = false

			const writeEvent = (event: string, data: unknown): void => {
				if (closed) return
				res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`)
			}

			// Send all currently pending prompts immediately
			for (const pending of prompt.pending()) {
				writeEvent('prompt', pending)
			}

			// Subscribe this connection to future events
			const unsubPending = prompt.onPending((pending) => writeEvent('prompt', pending))
			const unsubAnswered = prompt.onAnswered((id) => writeEvent('answered', { id }))
			const unsubExpired = prompt.onExpired((id) => writeEvent('expired', { id }))

			req.on('close', () => {
				closed = true
				sseConnections.delete(res)
				unsubPending()
				unsubAnswered()
				unsubExpired()
			})
		},
	})

	// POST endpoint: POST /api/prompts — submit a prompt response

		method: 'POST',
		path: '/api/prompts',
		async handler({ request: req, response: res }: RouteHandlerContext) {
			const bodyResult = await parseJsonBody(req, 1024 * 64)
			if (!bodyResult.success) {
				sendErrorResponse(res, 400, bodyResult.error.message)
				return
			}

			const body = bodyResult.value
			if (!isRecord(body) || typeof body.id !== 'string' || !('value' in body)) {
				sendErrorResponse(res, 422, 'Missing required fields: id, value')
				return
			}

			const answered = prompt.answer(body.id, body.value)
			if (answered) {
				sendJson(res, { success: true, id: body.id })
			} else {
				sendErrorResponse(res, 404, `Prompt not found or already answered: ${body.id}`)
			}
		},
	})

		// In SEA mode: run the executable directly. In dev mode: run via node.

		const executableArgs = sealed ? companionFlags : [companionScript, ...companionFlags]

		try {
			if (process.platform === 'win32') {

					detached: true,
					stdio: 'ignore',

			} else {
				const terminal = process.platform === 'darwin' ? 'open' : 'x-terminal-emulator'

					detached: true,
					stdio: 'ignore',
					cwd: projectRoot,
				}).unref()
			}
			logger.success('Prompt companion launched in a new terminal')
			return true
		} catch {
			logger.warn('Could not launch prompt companion')
			return false
		}
	}

			'- **fill**: Fill a template with values, substituting all placeholders.',
			"- **validate**: Validate fill values against a template's placeholders.",
			'- **templates**: List all registered templates.',
			'- **remove**: Remove a template by id.',
			'',
			'Remote companion operations:',
			'- **status**: Check if the prompt companion is connected. Returns connection info and the command to launch manually.',
			'- **launch**: Launch the prompt companion in a new terminal window. Will not launch if already connected.',
			'',
			'IMPORTANT — Before using any prompt operation (input, password, confirm, select, checkbox, editor),',
			'call **status** first to check if the companion is connected. If not connected, launch it using',
			"the IDE's built-in terminal by running the command from the status response (e.g. `run_in_terminal`).",
			'This keeps the companion inside the IDE where the user can see and interact with it directly.',
			'Only fall back to the **launch** operation if the IDE terminal is unavailable — it opens an',
			'external terminal window which may be less convenient.',
		].join('\n'),
		terminal: prompt,
		templates: promptTemplates,
		stores: promptTemplateStores,
		remote: {
			port: allocatedPort,
			token: serverToken,
			script: companionScript,
			sealed,
			connected: () => sseConnections.size > 0,
			launch: launchCompanion,
		},
	})

	// === MCP Server — Dual Transport (stdio + SSE)
	//
	// Two MCPServer instances share the same set of tools:
	// 1. Primary server on stdio — the IDE connects here (launched via mcp.json stdio entry)
	// 2. Secondary server on Streamable HTTP — exposed at /mcp on the HTTP server
	//    so additional clients can connect via the mcp.json SSE entry
	//
	// The stdio entry launches the process, which also starts the HTTP server.
	// Both transports serve the same tools with identical capabilities.

	const httpTransport = createHTTPServerTransport({
		sessionId: () => crypto.randomUUID(),
		hosts: ['localhost', '127.0.0.1', '[::1]'],
	})

	// Helper: read the raw request body as a string
	function readBody(req: IncomingMessage): Promise<string> {
		return new Promise((resolve, reject) => {
			let body = ''
			req.on('data', (chunk: Buffer) => {
				body += chunk.toString('utf-8')
			})
			req.on('end', () => resolve(body))
			req.on('error', reject)
		})
	}

	// POST /mcp — main JSON-RPC endpoint
	httpServer.routes.register({
		method: 'POST',
		path: '/mcp',
		async handler({ request: req, response: res }: RouteHandlerContext) {
			const body = await readBody(req)
			const webReq = toWebRequest(req, body)
			const webRes = await httpTransport.handle(webReq)
			await writeWebResponse(res, webRes)
		},
	})

	// GET /mcp — SSE stream for server-initiated messages
	httpServer.routes.register({
		method: 'GET',
		path: '/mcp',
		async handler({ request: req, response: res }: RouteHandlerContext) {
			const webReq = toWebRequest(req)
			const webRes = await httpTransport.handle(webReq)
			await writeWebResponse(res, webRes)
		},
	})

	// DELETE /mcp — session termination
	httpServer.routes.register({
		method: 'DELETE',
		path: '/mcp',
		async handler({ request: req, response: res }: RouteHandlerContext) {
			const webReq = toWebRequest(req)
			const webRes = await httpTransport.handle(webReq)
			await writeWebResponse(res, webRes)
		},
	})

	const mcpTools = [
		reasonTool,
		filesystemTool,
		interpretTool,
		agentTool,
		sandboxTool,
		promptTool,
		workflowTool,
	]

	const mcpDescription = [
		'Orkestrel provides a deterministic reasoning coprocessor, an in-memory filesystem, a natural language interpreter, an autonomous agent, an on-disk sandbox, interactive terminal prompts, and declarative workflow tracking.',
		'Use the `reason` tool whenever you need guaranteed-correct math, logic, symbolic algebra, or inference.',
		'Use the `filesystem` tool to safely load, edit, and persist files.',
		'Use the `interpret` tool to convert natural language into structured computation inputs.',
		'Use the `agent` tool to delegate tasks to a specialized sub-agent with autonomous execution. You can scope its tools, provide instructions, and attach documents.',
		'Use the `sandbox` tool to create isolated temporary directories for on-disk file operations with path-guarded safety.',
		'Use the `prompt` tool to interactively gather user input and manage reusable prompt templates. Before using prompt operations, check the status to see if the companion is connected. If not, launch it in the IDE terminal using the command from the status response (preferred) or fall back to the launch operation.',
		'Use the `workflow` tool to create, track, and manage workflows with phases and tasks. Workflows are declarative blueprints for organizing work.',
		'NEVER hallucinate computations — delegate to the reason tool instead.',
	].join('\n')

	// Primary MCP server — stdio transport for IDE communication
	const stdioServer = createMCPServer({
		name: 'orkestrel',
		version: '0.0.1',
		tools: mcpTools,
		description: mcpDescription,
	})

	// Secondary MCP server — Streamable HTTP transport for SSE clients
	const sseServer = createMCPServer({
		name: 'orkestrel',
		version: '0.0.1',
		transport: httpTransport,
		tools: mcpTools,
		description: mcpDescription,
	})

	// WebSocket upgrade handler — accepts ws:// connections on /mcp
	const wsUpgrade = createWebSocketUpgradeHandler(
		(socket) => {
			const transport = new WebSocketServerTransport(socket)
			const wsServer = createMCPServer({
				name: 'orkestrel',
				version: '0.0.1',
				transport,
				tools: mcpTools,
				description: mcpDescription,
			})
			wsServer.start()
			transport.onClose = () => wsServer.stop()
		},
		{ path: '/mcp' },
	)

	httpServer.http?.on('upgrade', wsUpgrade)

	// Initialize tools
	logger.debug('Initializing tools...')
	await reasonTool.init()
	await filesystemTool.init()
	await interpretTool.init()
	await promptTool.init()
	await workflowTool.init()
	logger.debug('Tools initialized')

	// Render startup banner to stderr (keeps stdout clean for JSON-RPC)
	renderBanner(output, {
		port: allocatedPort,
		tools: mcpTools,
	})

	// === Graceful Shutdown

	let shuttingDown = false

	function shutdown(): void {
		if (shuttingDown) return
		shuttingDown = true

		logger.info('Shutting down...')

		// Broadcast shutdown to all SSE clients so prompt companions exit cleanly
		broadcastShutdown()
		logger.debug(`Broadcast shutdown to ${sseConnections.size} SSE client(s)`)

		// Expire all pending prompts
		prompt.destroy()

		// Stop the HTTP server (graceful drain)
		httpServer.stop()

		// Stop both MCP servers
		stdioServer.stop()
		sseServer.stop()

		logger.success('Shutdown complete')

		// Allow a brief drain period before exiting
		setTimeout(() => process.exit(0), 500)
	}

	// MCP client disconnected (IDE closed the stdio pipe)
	process.stdin.on('end', shutdown)
	process.stdin.on('close', shutdown)

	// Manual termination
	process.on('SIGINT', shutdown)
	process.on('SIGTERM', shutdown)

	// Start both MCP servers
	logger.ready(
		`MCP on stdio + SSE at http://localhost:${allocatedPort}/mcp — prompt companion on /api/prompts`,
	)
	stdioServer.start()
	sseServer.start()
})()
