import { describe, it, expect } from 'vitest'
import { Seal } from '@orkestrel/seal'
import { Reporter } from '@orkestrel/core'

describe('Seal', () => {
	it('starts with idle status', () => {
		const seal = new Seal({
			name: 'test',
			entry: 'dist/bin/serve.cjs',
			output: 'dist/sea',
			reporter: new Reporter({ silent: true }),
		})

		expect(seal.status).toBe('idle')
	})

	it('fires onError for build failures', async () => {
		const errors: string[] = []

		const seal = new Seal({
			name: 'test',
			entry: 'nonexistent.cjs',
			output: 'dist/sea',
			reporter: new Reporter({ silent: true }),
			onError: (err: Error) => errors.push(err.message),
		})

		// This will fail because the entry doesn't exist or platform issues,
		// but we verify the lifecycle hooks work
		try {
			await seal.execute()
		} catch {
			// Expected
		}

		expect(seal.status).toBe('error')
	})

	it('subscription methods return unsubscribe functions', () => {
		const seal = new Seal({
			name: 'test',
			entry: 'test.cjs',
			output: 'dist/sea',
			reporter: new Reporter({ silent: true }),
		})

		const unsub1 = seal.onStep(() => {})
		const unsub2 = seal.onComplete(() => {})
		const unsub3 = seal.onError(() => {})

		expect(typeof unsub1).toBe('function')
		expect(typeof unsub2).toBe('function')
		expect(typeof unsub3).toBe('function')

		// Should not throw
		unsub1()
		unsub2()
		unsub3()
	})
})
