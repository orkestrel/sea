/**
 * Scripts test environment setup
 * Runs once before all script/build tests (Node environment)
 *
 * Provides shared factories, filesystem helpers, and lifecycle hooks
 * using REAL implementations. No mocks or fakes.
 */

import { beforeEach, afterEach, vi } from "vitest";
import { mkdirSync, writeFileSync, rmSync, existsSync } from "node:fs";
import { join } from "node:path";
import { tmpdir } from "node:os";

// === Lifecycle

const originalPlatform = process.platform;

beforeEach(() => {
  vi.spyOn(console, "error").mockImplementation(() => {});
  vi.spyOn(console, "log").mockImplementation(() => {});
  vi.spyOn(console, "warn").mockImplementation(() => {});
});

afterEach(() => {
  // Restore original platform after each test
  Object.defineProperty(process, "platform", {
    value: originalPlatform,
    writable: true,
    configurable: true,
  });
  vi.restoreAllMocks();
});

// === Temp Directory Factory

/**
 * A real temporary directory with cleanup.
 */
export interface TempDir {
  readonly path: string;
  readonly cleanup: () => void;
}

/**
 * Create a real temporary directory for testing.
 *
 * @param prefix - Prefix for the temp directory name
 * @returns TempDir with path and cleanup function
 */
export function createTempDir(prefix: string = "scripts-test"): TempDir {
  const path = join(tmpdir(), `${prefix}-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`);
  mkdirSync(path, { recursive: true });
  return {
    path,
    cleanup() {
      if (existsSync(path)) {
        rmSync(path, { recursive: true, force: true });
      }
    },
  };
}

// === File Factory

/**
 * Create test files in a directory.
 *
 * @param dir - Directory to create files in
 * @param files - Map of filename to content
 */
export function createTestFiles(dir: string, files: Record<string, string>): void {
  mkdirSync(dir, { recursive: true });
  for (const [name, content] of Object.entries(files)) {
    const filePath = join(dir, name);
    const parentDir = filePath.substring(0, filePath.lastIndexOf("/"));
    if (parentDir.length > 0) {
      mkdirSync(parentDir, { recursive: true });
    }
    writeFileSync(filePath, content);
  }
}

// === PE Factory

/**
 * Create a minimal valid PE executable buffer for testing.
 *
 * @param subsystem - PE subsystem value (default: 3 for console)
 * @returns Buffer containing a minimal PE executable
 */
export function createMinimalPe(subsystem: number = 3): Buffer {
  const buf = Buffer.alloc(256, 0);
  // DOS magic
  buf[0] = 0x4d;
  buf[1] = 0x5a;
  // PE offset at 0x3C → 0x80
  buf.writeUInt32LE(0x80, 0x3c);
  // PE signature at 0x80
  buf[0x80] = 0x50;
  buf[0x81] = 0x45;
  // PE32+ magic at peOffset + 0x18 = 0x98
  buf.writeUInt16LE(0x20b, 0x98);
  // Subsystem at peOffset + 0x5C = 0x80 + 0x5C = 0xDC
  buf.writeUInt16LE(subsystem, 0xdc);
  return buf;
}

/**
 * Create a minimal PE executable with a fake Authenticode signature.
 *
 * @param subsystem - PE subsystem value (default: 3 for console)
 * @returns Buffer containing a PE executable with a certificate table entry
 *
 * @remarks
 * The certificate data is appended at the end. The Certificate Table
 * data directory entry (index 4) points to it. Used to test
 * `stripPeSignature`.
 */
export function createSignedPe(subsystem: number = 3): Buffer {
  const certData = Buffer.from("FAKE_CERT_DATA_FOR_TESTING");
  const certOffset = 512;
  const buf = Buffer.alloc(certOffset + certData.length, 0);

  // DOS magic
  buf[0] = 0x4d;
  buf[1] = 0x5a;
  // PE offset at 0x3C → 0x80
  buf.writeUInt32LE(0x80, 0x3c);
  // PE signature at 0x80
  buf[0x80] = 0x50;
  buf[0x81] = 0x45;
  // PE32+ magic at peOffset + 0x18 = 0x98
  buf.writeUInt16LE(0x20b, 0x98);
  // Subsystem at peOffset + 0x5C = 0xDC
  buf.writeUInt16LE(subsystem, 0xdc);

  // Certificate Table is data directory entry index 4
  // For PE32+: data dir base = peOffset + 0x88 = 0x80 + 0x88 = 0x108
  // Cert entry = dataDirBase + 4 * 8 = 0x108 + 32 = 0x128
  const certEntryOffset = 0x128;
  buf.writeUInt32LE(certOffset, certEntryOffset); // RVA
  buf.writeUInt32LE(certData.length, certEntryOffset + 4); // Size

  // Append certificate data
  certData.copy(buf, certOffset);

  return buf;
}
