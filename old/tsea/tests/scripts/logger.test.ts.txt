import { describe, it, expect, vi, beforeEach, afterEach } from "vitest";
import { Logger } from "../../scripts/logger.js";

describe("Logger", () => {
  let stdoutSpy: ReturnType<typeof vi.spyOn>;

  beforeEach(() => {
    stdoutSpy = vi.spyOn(process.stdout, "write").mockImplementation(() => true);
  });

  afterEach(() => {
    stdoutSpy.mockRestore();
  });

  // Helper to get the last string written to stdout
  function lastOutput(): string {
    return String(stdoutSpy.mock.calls[stdoutSpy.mock.calls.length - 1]?.[0] ?? "");
  }

  describe("construction", () => {
    it("creates a logger with default options", () => {
      const log = new Logger();
      log.info("test");
      expect(stdoutSpy).toHaveBeenCalledOnce();
    });

    it("creates a silent logger that suppresses output", () => {
      const log = new Logger({ silent: true });
      log.info("test");
      log.success("test");
      log.warn("test");
      log.error("test");
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("info", () => {
    it("outputs message with blue info icon", () => {
      const log = new Logger();
      log.info("hello world");
      expect(stdoutSpy).toHaveBeenCalledOnce();
      expect(lastOutput()).toContain("ℹ");
      expect(lastOutput()).toContain("hello world");
    });
  });

  describe("success", () => {
    it("outputs message with green check icon", () => {
      const log = new Logger();
      log.success("done");
      expect(lastOutput()).toContain("✔");
      expect(lastOutput()).toContain("done");
    });
  });

  describe("warn", () => {
    it("outputs message with yellow warning icon", () => {
      const log = new Logger();
      log.warn("caution");
      expect(lastOutput()).toContain("⚠");
      expect(lastOutput()).toContain("caution");
    });
  });

  describe("error", () => {
    it("outputs message with red error icon", () => {
      const log = new Logger();
      log.error("failed");
      expect(lastOutput()).toContain("✖");
      expect(lastOutput()).toContain("failed");
    });
  });

  describe("step", () => {
    it("outputs numbered step with arrow icon", () => {
      const log = new Logger();
      log.step(3, 10, "doing something");
      expect(lastOutput()).toContain("→");
      expect(lastOutput()).toContain("[3/10]");
      expect(lastOutput()).toContain("doing something");
    });

    it("handles step 1 of 1", () => {
      const log = new Logger();
      log.step(1, 1, "only step");
      expect(lastOutput()).toContain("[1/1]");
      expect(lastOutput()).toContain("only step");
    });
  });

  describe("done", () => {
    it("outputs completion message with celebration icon", () => {
      const log = new Logger();
      log.done("all done");
      expect(lastOutput()).toContain("🎉");
      expect(lastOutput()).toContain("all done");
    });
  });

  describe("section", () => {
    it("outputs bold underlined section header", () => {
      const log = new Logger();
      log.section("Build Phase");
      expect(lastOutput()).toContain("Build Phase");
      expect(lastOutput()).toContain("\x1b[1m");
      expect(lastOutput()).toContain("\x1b[4m");
    });
  });

  describe("blank", () => {
    it("outputs an empty line", () => {
      const log = new Logger();
      log.blank();
      expect(lastOutput()).toBe("\n");
    });
  });

  describe("detail", () => {
    it("outputs indented detail with dot icon", () => {
      const log = new Logger();
      log.detail("some detail");
      expect(lastOutput()).toContain("·");
      expect(lastOutput()).toContain("some detail");
    });
  });

  describe("fatal", () => {
    it("outputs error message and throws", () => {
      const log = new Logger();
      expect(() => log.fatal("critical failure")).toThrow("critical failure");
      expect(lastOutput()).toContain("✖");
      expect(lastOutput()).toContain("critical failure");
    });

    it("throws even when silent", () => {
      const log = new Logger({ silent: true });
      expect(() => log.fatal("silent failure")).toThrow("silent failure");
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("debug", () => {
    it("outputs dimmed message with ellipsis icon", () => {
      const log = new Logger();
      log.debug("verbose detail");
      expect(lastOutput()).toContain("⋯");
      expect(lastOutput()).toContain("verbose detail");
    });
  });

  describe("skip", () => {
    it("outputs skip message with circle-slash icon", () => {
      const log = new Logger();
      log.skip("skipped step");
      expect(lastOutput()).toContain("⊘");
      expect(lastOutput()).toContain("skipped step");
    });
  });

  describe("timing", () => {
    it("outputs timing with stopwatch icon and milliseconds", () => {
      const log = new Logger();
      log.timing("Build", 42);
      expect(lastOutput()).toContain("⏱");
      expect(lastOutput()).toContain("Build");
      expect(lastOutput()).toContain("42ms");
    });

    it("formats seconds for values >= 1000", () => {
      const log = new Logger();
      log.timing("Deploy", 3456);
      expect(lastOutput()).toContain("3.46s");
    });
  });

  describe("pair", () => {
    it("outputs key-value pair with triangle icon", () => {
      const log = new Logger();
      log.pair("Version", "1.0.0");
      expect(lastOutput()).toContain("▸");
      expect(lastOutput()).toContain("Version");
      expect(lastOutput()).toContain("1.0.0");
    });
  });

  describe("list", () => {
    it("outputs each item with bullet icon", () => {
      const log = new Logger();
      log.list(["alpha", "bravo", "charlie"]);
      expect(stdoutSpy).toHaveBeenCalledTimes(3);
      const first = String(stdoutSpy.mock.calls[0]?.[0]);
      const second = String(stdoutSpy.mock.calls[1]?.[0]);
      const third = String(stdoutSpy.mock.calls[2]?.[0]);
      expect(first).toContain("•");
      expect(first).toContain("alpha");
      expect(second).toContain("bravo");
      expect(third).toContain("charlie");
    });

    it("outputs nothing for empty list", () => {
      const log = new Logger();
      log.list([]);
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("divider", () => {
    it("outputs a horizontal line", () => {
      const log = new Logger();
      log.divider();
      expect(lastOutput()).toContain("─");
    });
  });

  describe("progress", () => {
    it("renders a progress bar with filled and empty blocks", () => {
      const log = new Logger();
      log.progress(5, 10, "Halfway");
      expect(stdoutSpy).toHaveBeenCalledOnce();
      expect(lastOutput()).toContain("█");
      expect(lastOutput()).toContain("░");
      expect(lastOutput()).toContain("50%");
      expect(lastOutput()).toContain("Halfway");
    });

    it("handles 0% progress", () => {
      const log = new Logger();
      log.progress(0, 10);
      expect(lastOutput()).toContain("0%");
    });

    it("handles 100% progress", () => {
      const log = new Logger();
      log.progress(10, 10);
      expect(lastOutput()).toContain("100%");
    });

    it("clamps progress above 100%", () => {
      const log = new Logger();
      log.progress(15, 10);
      expect(lastOutput()).toContain("100%");
    });

    it("does nothing when silent", () => {
      const log = new Logger({ silent: true });
      log.progress(5, 10);
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("spinner", () => {
    it("returns a handle with stop and fail methods", () => {
      const log = new Logger({ silent: true });
      const handle = log.spinner("working...");
      expect(typeof handle.stop).toBe("function");
      expect(typeof handle.fail).toBe("function");
      handle.stop("done");
    });

    it("stop clears interval, logs success, and is idempotent", () => {
      vi.useFakeTimers();
      const log = new Logger();

      const handle = log.spinner("loading");
      vi.advanceTimersByTime(80);

      handle.stop("loaded!");
      expect(lastOutput()).toContain("✔");
      expect(lastOutput()).toContain("loaded!");

      const callCount = stdoutSpy.mock.calls.length;
      handle.stop("loaded again!");
      expect(stdoutSpy.mock.calls.length).toBe(callCount);

      vi.useRealTimers();
    });

    it("fail clears interval, logs error, and is idempotent", () => {
      vi.useFakeTimers();
      const log = new Logger();

      const handle = log.spinner("loading");
      vi.advanceTimersByTime(80);

      handle.fail("crashed!");
      expect(lastOutput()).toContain("✖");
      expect(lastOutput()).toContain("crashed!");

      const callCount = stdoutSpy.mock.calls.length;
      handle.fail("crashed again!");
      expect(stdoutSpy.mock.calls.length).toBe(callCount);

      vi.useRealTimers();
    });
  });

  describe("capture and release", () => {
    it("swallows console.log", () => {
      const log = new Logger();
      log.capture();

      const before = stdoutSpy.mock.calls.length;
      console.log("should be swallowed");
      expect(stdoutSpy.mock.calls.length).toBe(before);

      log.release();
    });

    it("swallows console.warn", () => {
      const log = new Logger();
      log.capture();

      const before = stdoutSpy.mock.calls.length;
      console.warn("should be swallowed");
      expect(stdoutSpy.mock.calls.length).toBe(before);

      log.release();
    });

    it("swallows console.error", () => {
      const log = new Logger();
      log.capture();

      const before = stdoutSpy.mock.calls.length;
      console.error("should be swallowed");
      expect(stdoutSpy.mock.calls.length).toBe(before);

      log.release();
    });

    it("swallows process.stdout.write", () => {
      const log = new Logger();
      log.capture();

      const before = stdoutSpy.mock.calls.length;
      process.stdout.write("should be swallowed\n");
      expect(stdoutSpy.mock.calls.length).toBe(before);

      log.release();
    });

    it("swallows process.stderr.write", () => {
      const stderrSpy = vi.spyOn(process.stderr, "write").mockImplementation(() => true);
      const log = new Logger();
      log.capture();

      const beforeStdout = stdoutSpy.mock.calls.length;
      const beforeStderr = stderrSpy.mock.calls.length;
      process.stderr.write("should be swallowed\n");
      expect(stdoutSpy.mock.calls.length).toBe(beforeStdout);
      expect(stderrSpy.mock.calls.length).toBe(beforeStderr);

      log.release();
      stderrSpy.mockRestore();
    });

    it("logger methods still produce output during capture", () => {
      const log = new Logger();
      log.capture();

      log.info("visible");
      expect(lastOutput()).toContain("ℹ");
      expect(lastOutput()).toContain("visible");

      log.success("also visible");
      expect(lastOutput()).toContain("✔");

      log.release();
    });

    it("propagates callback on swallowed stdout write", () => {
      const log = new Logger();
      log.capture();

      let called = false;
      process.stdout.write("swallowed\n", () => {
        called = true;
      });

      expect(called).toBe(true);

      log.release();
    });

    it("propagates callback with encoding on swallowed write", () => {
      const log = new Logger();
      log.capture();

      let called = false;
      process.stdout.write("swallowed\n", "utf8", () => {
        called = true;
      });

      expect(called).toBe(true);

      log.release();
    });

    it("restores console.log on release", () => {
      const originalLog = console.log;
      const log = new Logger();
      log.capture();
      log.release();

      expect(console.log).toBe(originalLog);
    });

    it("restores console.warn on release", () => {
      const originalWarn = console.warn;
      const log = new Logger();
      log.capture();
      log.release();

      expect(console.warn).toBe(originalWarn);
    });

    it("restores console.error on release", () => {
      const originalError = console.error;
      const log = new Logger();
      log.capture();
      log.release();

      expect(console.error).toBe(originalError);
    });

    it("release is safe to call without prior capture", () => {
      const log = new Logger();
      log.release();
      log.info("still works");
      expect(stdoutSpy.mock.calls.length).toBeGreaterThanOrEqual(1);
    });
  });

  describe("silent mode", () => {
    it("suppresses all output methods", () => {
      const log = new Logger({ silent: true });
      log.info("a");
      log.success("b");
      log.warn("c");
      log.error("d");
      log.debug("e");
      log.skip("f");
      log.timing("g", 100);
      log.step(1, 2, "h");
      log.done("i");
      log.section("j");
      log.blank();
      log.detail("k");
      log.pair("l", "m");
      log.list(["n", "o"]);
      log.divider();
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("level filtering", () => {
    it("defaults to debug level — all messages pass", () => {
      const log = new Logger();
      log.debug("a");
      log.info("b");
      log.warn("c");
      log.error("d");
      expect(stdoutSpy).toHaveBeenCalledTimes(4);
    });

    it("level info suppresses debug and skip", () => {
      const log = new Logger({ level: "info" });
      log.debug("suppressed");
      log.skip("suppressed");
      expect(stdoutSpy).not.toHaveBeenCalled();

      log.info("visible");
      log.success("visible");
      log.warn("visible");
      log.error("visible");
      expect(stdoutSpy).toHaveBeenCalledTimes(4);
    });

    it("level warn suppresses debug, info, and related methods", () => {
      const log = new Logger({ level: "warn" });
      log.debug("suppressed");
      log.info("suppressed");
      log.success("suppressed");
      log.step(1, 2, "suppressed");
      log.detail("suppressed");
      log.pair("key", "val");
      log.list(["a", "b"]);
      log.timing("t", 100);
      log.divider();
      log.section("suppressed");
      log.done("suppressed");
      log.blank();
      log.skip("suppressed");
      expect(stdoutSpy).not.toHaveBeenCalled();

      log.warn("visible");
      log.error("visible");
      expect(stdoutSpy).toHaveBeenCalledTimes(2);
    });

    it("level error suppresses everything except error and fatal", () => {
      const log = new Logger({ level: "error" });
      log.debug("suppressed");
      log.info("suppressed");
      log.warn("suppressed");
      log.success("suppressed");
      expect(stdoutSpy).not.toHaveBeenCalled();

      log.error("visible");
      expect(stdoutSpy).toHaveBeenCalledOnce();
    });

    it("level error allows fatal to output before throwing", () => {
      const log = new Logger({ level: "error" });
      expect(() => log.fatal("crash")).toThrow("crash");
      expect(lastOutput()).toContain("crash");
    });

    it("level warn suppresses progress bar", () => {
      const log = new Logger({ level: "warn" });
      log.progress(5, 10);
      expect(stdoutSpy).not.toHaveBeenCalled();
    });

    it("level warn returns noop spinner", () => {
      const log = new Logger({ level: "warn" });
      const handle = log.spinner("working");
      handle.stop("done");
      // Spinner is a noop — no output
      expect(stdoutSpy).not.toHaveBeenCalled();
    });
  });

  describe("integration", () => {
    it("logs a multi-step build sequence", () => {
      const log = new Logger();
      log.section("Build");
      log.step(1, 3, "Prepare");
      log.success("Ready");
      log.step(2, 3, "Compile");
      log.success("Compiled");
      log.detail("Output: dist/main.js");
      log.pair("Size", "42kb");
      log.step(3, 3, "Package");
      log.success("Packaged");
      log.timing("Total", 1234);
      log.done("Complete!");
      log.blank();
      expect(stdoutSpy).toHaveBeenCalledTimes(12);
    });

    it("capture swallows external output while logger output flows", () => {
      const log = new Logger();
      log.capture();

      // External output is swallowed
      console.log("swallowed 1");
      console.warn("swallowed 2");
      console.error("swallowed 3");

      // Logger output flows through
      log.step(1, 2, "Working");
      expect(lastOutput()).toContain("Working");

      log.success("Done");
      expect(lastOutput()).toContain("✔");

      // Only 2 logger calls should have written to stdout
      expect(stdoutSpy).toHaveBeenCalledTimes(2);

      log.release();
    });
  });
});
