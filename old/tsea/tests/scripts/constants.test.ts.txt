import { describe, it, expect } from "vitest";
import {
  ROOT_DIR,
  DIST_DIR,
  DIST_CLIENT_DIR,
  DIST_SERVER_DIR,
  SEA_CONFIG_PATH,
  SEA_BLOB_PATH,
  SEA_SENTINEL_FUSE,
  SEA_BLOB_RESOURCE,
  DEFAULT_COMPRESSION_QUALITY,
  PLATFORMS,
  PE_SUBSYSTEM_CONSOLE,
  PE_SUBSYSTEM_WINDOWS,
} from "../../scripts/constants.js";
import { existsSync } from "node:fs";
import { resolve, join } from "node:path";

describe("scripts/constants", () => {
  describe("build paths", () => {
    it("should resolve ROOT_DIR to current working directory", () => {
      expect(ROOT_DIR).toBe(resolve("."));
    });

    it("should place DIST_DIR under ROOT_DIR", () => {
      expect(DIST_DIR).toBe(join(ROOT_DIR, "dist"));
    });

    it("should place DIST_CLIENT_DIR under DIST_DIR", () => {
      expect(DIST_CLIENT_DIR).toBe(join(DIST_DIR, "client"));
    });

    it("should place DIST_SERVER_DIR under DIST_DIR", () => {
      expect(DIST_SERVER_DIR).toBe(join(DIST_DIR, "server"));
    });

    it("should point SEA_CONFIG_PATH to configs folder", () => {
      expect(SEA_CONFIG_PATH).toContain("configs");
      expect(SEA_CONFIG_PATH).toContain("sea-config.json");
    });

    it("should point SEA_BLOB_PATH to dist folder", () => {
      expect(SEA_BLOB_PATH).toBe(join(DIST_DIR, "sea-prep.blob"));
    });

    it("should have SEA config file present", () => {
      expect(existsSync(SEA_CONFIG_PATH)).toBe(true);
    });
  });

  describe("SEA constants", () => {
    it("should define SEA_SENTINEL_FUSE as non-empty string", () => {
      expect(typeof SEA_SENTINEL_FUSE).toBe("string");
      expect(SEA_SENTINEL_FUSE.length).toBeGreaterThan(0);
      expect(SEA_SENTINEL_FUSE).toContain("NODE_SEA_FUSE");
    });

    it("should define SEA_BLOB_RESOURCE as NODE_SEA_BLOB", () => {
      expect(SEA_BLOB_RESOURCE).toBe("NODE_SEA_BLOB");
    });
  });

  describe("compression constants", () => {
    it("should set DEFAULT_COMPRESSION_QUALITY to 11 (max Brotli)", () => {
      expect(DEFAULT_COMPRESSION_QUALITY).toBe(11);
    });
  });

  describe("PLATFORMS", () => {
    it("should be a record indexed by NodeJS.Platform", () => {
      expect(typeof PLATFORMS).toBe("object");
      expect(PLATFORMS).not.toBeNull();
    });

    it("should contain all major platforms", () => {
      const majorPlatforms: NodeJS.Platform[] = ["win32", "darwin", "linux"];

      majorPlatforms.forEach((name) => {
        const config = PLATFORMS[name];
        expect(config).toBeDefined();
        expect(config).toHaveProperty("executable");
        expect(config).toHaveProperty("removeSignature");
        expect(config).toHaveProperty("resign");
        expect(config).toHaveProperty("postjectArgs");
      });
    });

    it("should use .exe extension only for Windows-like platforms", () => {
      expect(PLATFORMS.win32.executable).toMatch(/\.exe$/);
      expect(PLATFORMS.cygwin.executable).toMatch(/\.exe$/);
      expect(PLATFORMS.darwin.executable).not.toMatch(/\.exe$/);
      expect(PLATFORMS.linux.executable).not.toMatch(/\.exe$/);
    });

    it("should enable signature removal for darwin and win32", () => {
      expect(PLATFORMS.darwin.removeSignature).toBe(true);
      expect(PLATFORMS.darwin.resign).toBe(true);
      expect(PLATFORMS.win32.removeSignature).toBe(true);
      expect(PLATFORMS.win32.resign).toBe(false);
      expect(PLATFORMS.linux.removeSignature).toBe(false);
      expect(PLATFORMS.linux.resign).toBe(false);
    });

    it("should only have postject args for darwin", () => {
      expect(PLATFORMS.darwin.postjectArgs.length).toBeGreaterThan(0);
      expect(PLATFORMS.win32.postjectArgs.length).toBe(0);
      expect(PLATFORMS.linux.postjectArgs.length).toBe(0);
    });

    it("should have frozen postjectArgs arrays", () => {
      for (const config of Object.values(PLATFORMS)) {
        expect(Object.isFrozen(config.postjectArgs)).toBe(true);
      }
    });

    it("should have all PlatformConfig fields on each entry", () => {
      for (const config of Object.values(PLATFORMS)) {
        expect(typeof config.executable).toBe("string");
        expect(typeof config.removeSignature).toBe("boolean");
        expect(typeof config.resign).toBe("boolean");
        expect(Array.isArray(config.postjectArgs)).toBe(true);
      }
    });

    it("should contain all 11 Node.js platforms", () => {
      const expectedPlatforms: NodeJS.Platform[] = [
        "win32",
        "darwin",
        "linux",
        "aix",
        "android",
        "cygwin",
        "freebsd",
        "haiku",
        "netbsd",
        "openbsd",
        "sunos",
      ];

      for (const name of expectedPlatforms) {
        expect(PLATFORMS[name]).toBeDefined();
      }
    });
  });

  describe("PE subsystem", () => {
    it("should define console subsystem as 3", () => {
      expect(PE_SUBSYSTEM_CONSOLE).toBe(3);
    });

    it("should define windows subsystem as 2", () => {
      expect(PE_SUBSYSTEM_WINDOWS).toBe(2);
    });

    it("should have different values for console and windows", () => {
      expect(PE_SUBSYSTEM_CONSOLE).not.toBe(PE_SUBSYSTEM_WINDOWS);
    });
  });
});
