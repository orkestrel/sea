import type { LoggerInterface, LoggerOptions, LogLevel, SpinnerHandle } from "./types.js";

// === ANSI escape codes

const RESET = "\x1b[0m";
const BOLD = "\x1b[1m";
const DIM = "\x1b[2m";
const UNDERLINE = "\x1b[4m";
const RED = "\x1b[31m";
const GREEN = "\x1b[32m";
const YELLOW = "\x1b[33m";
const BLUE = "\x1b[34m";
const MAGENTA = "\x1b[35m";
const CYAN = "\x1b[36m";
const WHITE = "\x1b[37m";
const GRAY = "\x1b[90m";

// === Level severity (higher = more severe)

const LEVEL_SEVERITY: Readonly<Record<LogLevel, number>> = {
  debug: 0,
  info: 1,
  warn: 2,
  error: 3,
};

// === Icons

const ICON_INFO = "ℹ";
const ICON_SUCCESS = "✔";
const ICON_WARN = "⚠";
const ICON_ERROR = "✖";
const ICON_STEP = "→";
const ICON_DONE = "🎉";
const ICON_DETAIL = "·";
const ICON_DEBUG = "⋯";
const ICON_SKIP = "⊘";
const ICON_TIMING = "⏱";
const ICON_BULLET = "•";
const ICON_PAIR = "▸";

// === Spinner frames

const SPINNER_FRAMES = ["⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"];
const SPINNER_INTERVAL = 80;

// === Progress bar

const BAR_FILLED = "█";
const BAR_EMPTY = "░";
const BAR_WIDTH = 30;

// === No-op sinks

/** No-op write for swallowing stream writes */
function noopWrite(chunk: Uint8Array | string, callback?: (err?: Error) => void): boolean;
function noopWrite(
  chunk: Uint8Array | string,
  encoding?: BufferEncoding,
  callback?: (err?: Error) => void,
): boolean;
function noopWrite(
  _chunk: Uint8Array | string,
  _encodingOrCb?: BufferEncoding | ((err?: Error) => void),
  _callback?: (err?: Error) => void,
): boolean {
  const cb = typeof _encodingOrCb === "function" ? _encodingOrCb : _callback;
  if (cb !== undefined) cb();
  return true;
}

/**
 * Semantic console logger with icons, ANSI colors, and interactive elements.
 *
 * @remarks
 * Each method maps to a semantic intent with a corresponding
 * unicode icon and ANSI color. When `silent` is true, all output
 * is suppressed (useful for testing).
 *
 * `capture()` silences all external output by replacing
 * `console.log`, `console.warn`, `console.error`,
 * `process.stdout.write`, and `process.stderr.write` with
 * no-ops. Only explicit logger method calls produce output.
 * Call `release()` to restore originals.
 *
 * @example
 * ```ts
 * const log = new Logger()
 * log.capture()
 * log.section('Building Application')
 * log.step(1, 5, 'Preparing directories...')
 * log.success('Directories ready')
 * log.done('Build complete!')
 * log.release()
 * ```
 */
export class Logger implements LoggerInterface {
  #silent: boolean;
  #level: number;
  #originalLog: typeof console.log | undefined;
  #originalWarn: typeof console.warn | undefined;
  #originalError: typeof console.error | undefined;
  #originalStdoutWrite: typeof process.stdout.write | undefined;
  #originalStderrWrite: typeof process.stderr.write | undefined;

  constructor(options: LoggerOptions = {}) {
    this.#silent = options.silent ?? false;
    this.#level = LEVEL_SEVERITY[options.level ?? "debug"];
  }

  info(message: string): void {
    this.#emit("info", `${BLUE}${ICON_INFO}${RESET} ${message}`);
  }

  success(message: string): void {
    this.#emit("info", `${GREEN}${ICON_SUCCESS}${RESET} ${message}`);
  }

  warn(message: string): void {
    this.#emit("warn", `${YELLOW}${ICON_WARN}${RESET} ${message}`);
  }

  error(message: string): void {
    this.#emit("error", `${RED}${ICON_ERROR}${RESET} ${message}`);
  }

  debug(message: string): void {
    this.#emit("debug", `${GRAY}${ICON_DEBUG} ${message}${RESET}`);
  }

  skip(message: string): void {
    this.#emit("debug", `${YELLOW}${DIM}${ICON_SKIP}${RESET} ${DIM}${message}${RESET}`);
  }

  timing(label: string, ms: number): void {
    const formatted = ms >= 1000 ? `${(ms / 1000).toFixed(2)}s` : `${String(Math.round(ms))}ms`;
    this.#emit("info", `${CYAN}${ICON_TIMING}${RESET} ${label} ${DIM}${formatted}${RESET}`);
  }

  step(current: number, total: number, message: string): void {
    this.#emit(
      "info",
      `${CYAN}${ICON_STEP}${RESET} ` +
        `${DIM}[${String(current)}/${String(total)}]${RESET} ` +
        message,
    );
  }

  done(message: string): void {
    this.#emit("info", `\n${GREEN}${BOLD}${ICON_DONE} ${message}${RESET}`);
  }

  section(title: string): void {
    this.#emit("info", `\n${BOLD}${UNDERLINE}${title}${RESET}\n`);
  }

  blank(): void {
    this.#emit("info", "");
  }

  detail(message: string): void {
    this.#emit("info", `   ${DIM}${ICON_DETAIL} ${message}${RESET}`);
  }

  pair(key: string, value: string): void {
    this.#emit(
      "info",
      `   ${MAGENTA}${ICON_PAIR}${RESET} ${WHITE}${key}${RESET} ${DIM}${value}${RESET}`,
    );
  }

  list(items: readonly string[]): void {
    for (const item of items) {
      this.#emit("info", `   ${CYAN}${ICON_BULLET}${RESET} ${item}`);
    }
  }

  divider(): void {
    this.#emit("info", `${DIM}${"─".repeat(50)}${RESET}`);
  }

  fatal(message: string): never {
    this.#emit("error", `${RED}${BOLD}${ICON_ERROR} ${message}${RESET}`);
    throw new Error(message);
  }

  progress(current: number, total: number, message?: string): void {
    if (this.#silent || LEVEL_SEVERITY.info < this.#level) return;
    const ratio = total > 0 ? Math.min(current / total, 1) : 0;
    const filled = Math.round(BAR_WIDTH * ratio);
    const empty = BAR_WIDTH - filled;
    const percent = `${String(Math.round(ratio * 100))}%`;
    const bar = `${CYAN}${BAR_FILLED.repeat(filled)}${GRAY}${BAR_EMPTY.repeat(empty)}${RESET}`;
    const label = message !== undefined ? ` ${message}` : "";
    const line = `  ${bar} ${DIM}${percent}${RESET}${label}`;
    this.#overwrite(line);
  }

  spinner(message: string): SpinnerHandle {
    if (this.#silent || LEVEL_SEVERITY.info < this.#level) {
      return {
        stop: () => {},
        fail: () => {},
      };
    }

    let frame = 0;
    let stopped = false;

    const timer = setInterval(() => {
      const icon = SPINNER_FRAMES[frame % SPINNER_FRAMES.length];
      this.#overwrite(`${CYAN}${icon ?? "⠋"}${RESET} ${message}`);
      frame++;
    }, SPINNER_INTERVAL);

    return {
      stop: (doneMessage: string) => {
        if (stopped) return;
        stopped = true;
        clearInterval(timer);
        this.#clearLine();
        this.success(doneMessage);
      },
      fail: (errorMessage: string) => {
        if (stopped) return;
        stopped = true;
        clearInterval(timer);
        this.#clearLine();
        this.error(errorMessage);
      },
    };
  }

  capture(): void {
    this.#originalLog = console.log;
    this.#originalWarn = console.warn;
    this.#originalError = console.error;
    this.#originalStdoutWrite = process.stdout.write.bind(process.stdout);
    this.#originalStderrWrite = process.stderr.write.bind(process.stderr);

    // Replace with no-ops — external code is silenced
    console.log = () => {};
    console.warn = () => {};
    console.error = () => {};
    process.stdout.write = noopWrite;
    process.stderr.write = noopWrite;
  }

  release(): void {
    if (this.#originalLog !== undefined) {
      console.log = this.#originalLog;
      this.#originalLog = undefined;
    }
    if (this.#originalWarn !== undefined) {
      console.warn = this.#originalWarn;
      this.#originalWarn = undefined;
    }
    if (this.#originalError !== undefined) {
      console.error = this.#originalError;
      this.#originalError = undefined;
    }
    if (this.#originalStdoutWrite !== undefined) {
      process.stdout.write = this.#originalStdoutWrite;
      this.#originalStdoutWrite = undefined;
    }
    if (this.#originalStderrWrite !== undefined) {
      process.stderr.write = this.#originalStderrWrite;
      this.#originalStderrWrite = undefined;
    }
  }

  // Write only if the message level meets the minimum threshold
  #emit(level: LogLevel, line: string): void {
    if (LEVEL_SEVERITY[level] >= this.#level) {
      this.#write(line);
    }
  }

  #write(line: string): void {
    if (this.#silent) return;
    // Always write to the original stdout, bypassing any capture
    const write = this.#originalStdoutWrite ?? process.stdout.write.bind(process.stdout);
    write(`${line}\n`);
  }

  #overwrite(line: string): void {
    if (this.#silent) return;
    const write = this.#originalStdoutWrite ?? process.stdout.write.bind(process.stdout);
    write(`\r\x1b[K${line}`);
  }

  #clearLine(): void {
    if (this.#silent) return;
    const write = this.#originalStdoutWrite ?? process.stdout.write.bind(process.stdout);
    write("\r\x1b[K");
  }
}
